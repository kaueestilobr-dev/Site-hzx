document.addEventListener('DOMContentLoaded', () => { // Captura dos inputs const channelSelect = document.getElementById('channelSelect'); const titleInput = document.getElementById('title'); const descriptionInput = document.getElementById('description'); const colorInput = document.getElementById('color'); const authorInput = document.getElementById('author'); const footerInput = document.getElementById('footer'); const imageInput = document.getElementById('image'); const sendBtn = document.getElementById('sendEmbed'); // Captura do visual (A Estilização / Preview) const previewTitle = document.getElementById('previewTitle'); const previewDescription = document.getElementById('previewDescription'); const embedBar = document.getElementById('embedBar'); const previewAuthor = document.getElementById('previewAuthor'); const previewFooter = document.getElementById('previewFooter'); const previewImage = document.getElementById('previewImage'); // ========================================== // 1. ESTILIZAÇÃO E PREVIEW EM TEMPO REAL // ========================================== titleInput.addEventListener('input', () => previewTitle.innerText = titleInput.value || 'Título'); descriptionInput.addEventListener('input', () => previewDescription.innerText = descriptionInput.value || 'Descrição da embed...'); colorInput.addEventListener('input', () => embedBar.style.backgroundColor = colorInput.value); authorInput.addEventListener('input', () => previewAuthor.innerText = authorInput.value || 'Autor'); footerInput.addEventListener('input', () => previewFooter.innerText = footerInput.value || 'HzX Vision'); imageInput.addEventListener('input', () => { if (imageInput.value.trim() !== '') { previewImage.src = imageInput.value; previewImage.style.display = 'block'; } else { previewImage.style.display = 'none'; } }); // ========================================== // 2. BUSCAR CANAIS DO RENDER // ========================================== fetch('/api/channels') .then(res => res.json()) .then(channels => { channelSelect.innerHTML = ''; channels.forEach(ch => { const opt = document.createElement('option'); opt.value = ch.id; opt.innerText = ch.name; channelSelect.appendChild(opt); }); }) .catch(err => console.error('Erro ao buscar canais:', err)); // ========================================== // 3. ENVIAR PARA O DISCORD // ========================================== sendBtn.addEventListener('click', async () => { const channelId = channelSelect.value; if (!channelId) { alert('⚠️ Selecione um canal primeiro!'); return; } const dadosEmbed = { channelId: channelId, title: titleInput.value, description: descriptionInput.value, color: colorInput.value, author: authorInput.value, footer: footerInput.value, image: imageInput.value }; try { const response = await fetch('/api/enviar-discord', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(dadosEmbed) }); const resultado = await response.json(); if (resultado.success) { alert('🚀 Embed enviada com sucesso!'); } else { alert('❌ Erro: ' + resultado.message); } } catch (error) { alert('❌ Erro ao conectar com o servidor.'); } }); });
